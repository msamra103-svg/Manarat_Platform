  // ------------------------------------------------------------------ camera (critically-damped follow, no shake)
  function updateCamera(dt) {
    const E = S.engine; if (!E) return; const p = E.player;
    const px = p.x, py = p.y;
    const moving = Math.abs(p.vx) > 1.2; const lookT = moving ? Math.sign(p.vx) * 1.6 : cam.look * 0.98;
    cam.look += (lookT - cam.look) * (1 - Math.exp(-dt * 1.4));
    if (p.onGround) cam.baseY += (py - cam.baseY) * (1 - Math.exp(-dt * 3.5));
    else if (py > cam.baseY + 3.2) cam.baseY += (py - 3.2 - cam.baseY) * (1 - Math.exp(-dt * 5));
    else if (py < cam.baseY - 0.4) cam.baseY += (py + 0.4 - cam.baseY) * (1 - Math.exp(-dt * 6));
    const halfW = Math.tan(camera.fov * Math.PI / 360) * CAM_Z * camera.aspect;
    cam.tx = clamp(px + cam.look, halfW - 0.5, E.W - halfW + 0.5); cam.ty = Math.max(cam.baseY, -1) + CAM_H + 0.6;
    const k = 1 - Math.exp(-dt * 6.5); cam.x += (cam.tx - cam.x) * k; cam.y += (cam.ty - cam.y) * k;
    camera.position.set(cam.x, cam.y + 1.2, CAM_Z); camera.lookAt(cam.x, cam.y - 0.25, 0);
  }

  // ------------------------------------------------------------------ main update
  function animState(E) {
    const p = E.player;
    if (S.phase === 'goal' || S.phase === 'stageClear' || S.phase === 'final') return { state: 'victory', speed: 0, vy: 0, facing: 1 };
    if (p.dead) return { state: S.dieCause === 'pit' ? 'pitfall' : 'hurt', speed: Math.abs(p.vx), vy: p.vy, hurtT: 0.1, facing: p.facing };
    if (p.hurtT > 0) return { state: 'hurt', speed: Math.abs(p.vx), vy: p.vy, hurtT: 0.42 - p.hurtT, facing: p.facing };
    if (!p.onGround) return { state: 'air', speed: Math.abs(p.vx), vy: p.vy, tuck: p.tuck, facing: p.facing };
    if (p.skidT > 0.03) return { state: 'skid', speed: Math.abs(p.vx), skidT: clamp(p.skidT / 0.3, 0, 1), facing: -Math.sign(p.vx) || p.facing };
    if (p.landT < 0.22 && p.landK > 0.3) return { state: 'land', speed: Math.abs(p.vx), landT: p.landT, landK: p.landK, facing: p.facing };
    const lookUp = E.nearSlot && Math.abs(E.nearSlot.dist) < 1.4 && E.nearSlot.under ? 1 : 0;
    if (Math.abs(p.vx) < 0.05) { const ledge = edgeNear(E, p); return { state: ledge ? 'balance' : 'idle', speed: 0, lookUp, facing: p.facing }; }
    return { state: 'run', speed: Math.abs(p.vx), accel: p.accel, lookUp, facing: p.facing };
  }
  function edgeNear(E, p) { const fx = p.x + p.facing * (p.w / 2 + 0.05), iy = Math.floor(p.y - 0.5); return !p.ground && !E.solid(Math.floor(fx), iy) && !E.oneWay(Math.floor(fx), iy) && E.solid(Math.floor(p.x), iy); }
  function update(dt) {
    S.time += dt; S.pt += dt; const E = S.engine;
    if (toastUntil && S.time > toastUntil) { $('toast').hidden = true; toastUntil = 0; }
    if (S.paused) return;
    readInput();
    const ph = S.phase;
    if (E) {
      if (ph === 'stageIntro') { E.input = { left: false, right: false, jump: false }; if (S.pt > (S.platform.on ? 0.45 : 2.0)) { hide('stageCard'); phase('play'); } }
      if (ph === 'play') { E.input = Object.assign({}, input); }
      if (ph === 'goal' || ph === 'dying' || ph === 'gameOver' || ph === 'stageClear') E.input = { left: false, right: false, jump: false };
      if (ph === 'menu') { showcaseAI(E, dt); }
      if (['play', 'stageIntro', 'menu', 'goal', 'dying'].includes(ph)) E.update(dt);
      if (S.stage) S.stage.t += ph === 'play' ? dt : 0;
      // dying: splash when the hero meets the water, then lose a life
      if (ph === 'dying') {
        const p = E.player; if (S.dieCause === 'pit' && !S.splashed && p.y < -0.65) { S.splashed = true; view.fx('splash', { x: p.x, y: -0.6 }); audio.splash(); }
        if (S.dieCause === 'pit') { p.y = Math.max(p.y - 6 * dt, -3.2); } else { p.vy -= 24 * dt; p.y += p.vy * dt; }
        if (S.pt > 1.6) afterDeath();
      }
      if (ph === 'goal') { S.goalRaise = Math.min(1, S.goalRaise + dt / 2.4); if (S.pt > (S.platform.on ? 0.65 : 4.4)) stageClear(); }
      if (ph === 'stageClear' && S.platform.on && S.pt > 0.65) nextStage();
      if (ph === 'question') {
        if (S.closeAt > 0 && S.time >= S.closeAt) closeQuestion();
        else if (cfg.timer && S.q && !S.q.locked && !S.platform.on) {
          S.qTimer -= dt; const pct = clamp(S.qTimer / cfg.timer, 0, 1); $('qTimerFill').style.width = (pct * 100).toFixed(1) + '%'; $('qTimer').classList.toggle('urgent', S.qTimer < 5.5); $('qTimerTxt').textContent = AR(Math.max(0, Math.ceil(S.qTimer)));
          if (S.qTimer <= 0) { S.q.tries++; S.wrongTotal++; S.results.push({ q: S.q.qIndex, stage: S.stage.id, try: S.q.tries, correct: false, timeout: true }); $('qFeedback').textContent = 'انتهى الوقت — حاول مرة أخرى'; $('qFeedback').className = 'q-feedback info'; audio.wrong(); S.qTimer = cfg.timer; const card = doc.querySelector('.q-card'); card.classList.remove('shake'); void card.offsetWidth; card.classList.add('shake'); }
        }
      }
      // guidance bubble at a closed gate
      if (ph === 'play') {
        const need = E.gatePushT > 0.35 || (E.nearSlot && E.nearSlot.dist < 3 && E.nearSlot.dist > -1.5 && S.stage.t > 2 && E.player.onGround && Math.abs(E.player.vx) < 0.3);
        if (need && $('hintBubble').hidden) { $('hintBubble').hidden = false; if (S.time - S.lastHint > 4) { audio.hint(); S.lastHint = S.time; } }
        else if (!need && !$('hintBubble').hidden && (!E.nearSlot || E.nearSlot.dist > 4)) $('hintBubble').hidden = true;
      } else if (!$('hintBubble').hidden) $('hintBubble').hidden = true;
      view.sync(dt, E, S);
      if (view.boy) { const st = animState(E); if (S.phase === 'menu' || ph === 'stageIntro') st.state = st.state === 'run' ? 'run' : st.state; view.syncBoy(dt, E, st); }
      // skid dust
      if (E.player.skidT > 0.03 && E.player.onGround && (S.time * 30 | 0) % 2 === 0) view.fx('skid', { x: E.player.x, y: E.player.y });
      updateCamera(dt);
      audio.update(dt, { speed: Math.abs(E.player.vx), theme: view.theme ? view.theme.id : 'corniche', paused: ph === 'question' });
      if (ph === 'play' || ph === 'goal') hud();
    }
  }
  // menu showcase: the hero strolls back and forth and hops now and then
  function showcaseAI(E, dt) {
    const p = E.player; S.sc = S.sc || { t: 0, dir: 1 }; S.sc.t += dt;
    if (p.x > 15) S.sc.dir = -1; if (p.x < 4) S.sc.dir = 1;
    const walk = (S.sc.t % 9) > 3.5; E.input = { left: walk && S.sc.dir < 0, right: walk && S.sc.dir > 0, jump: walk && (S.sc.t % 9) > 7.4 && (S.sc.t % 9) < 7.7 };
  }

  // ------------------------------------------------------------------ frame loop with auto-quality for slow devices / smart boards
  let last = performance.now();
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000); last = now;
    update(dt); view.render(camera);
    // auto-quality: sustained low frame rate → lower resolution, then post-processing, then shadows
    const pf = S.perf; pf.frames++; pf.fpsAvg = lerp(pf.fpsAvg, 1 / Math.max(dt, 1e-3), 0.05);
    if (pf.frames > 120 && S.phase === 'play') { if (pf.fpsAvg < 34) pf.slow += dt; else pf.slow = Math.max(0, pf.slow - dt * 0.5); if (pf.slow > 3 && pf.downgrades < 4) { pf.slow = 0; pf.downgrades++; degrade(pf.downgrades); } }
    requestAnimationFrame(frame);
  }
  function degrade(level) {
    const pf = S.perf;
    if (level === 1 || level === 3) { pf.dpr = Math.max(0.6, pf.dpr * 0.75); renderer.setPixelRatio(pf.dpr); resize(); }
    if (level === 2) view.world.postOK = false;
    if (level === 4) { renderer.shadowMap.enabled = false; view.world.sun.castShadow = false; view.boy && (view.boy.quality = 0); }
  }

  $('btnStart').addEventListener('click', start);
  $('btnHow').addEventListener('click', () => { audio.init(); show('how'); });
  doc.querySelectorAll('[data-close]').forEach(b => b.addEventListener('click', () => hide(b.dataset.close)));
  $('btnTeacher').addEventListener('click', () => { if (root.QHTeacher) root.QHTeacher.open(); });
  $('btnPause').addEventListener('click', () => togglePause());
  $('btnResume').addEventListener('click', () => togglePause(false));
  $('btnRestartStage').addEventListener('click', () => { togglePause(false); S.lives = Math.max(S.lives, 1); if (S.q) { hide('question'); S.q = null; S.closeAt = -1; } beginStage(S.stageIdx); });
  $('btnMenu').addEventListener('click', () => { togglePause(false); if (S.q) { hide('question'); S.q = null; } openMenu(); });
  $('btnFull').addEventListener('click', toggleFull);
  $('btnMute').addEventListener('click', () => { audio.init(); toggleMute(); });
  $('volMusic').addEventListener('input', e => audio.set('music', Number(e.target.value)));
  $('volSfx').addEventListener('input', e => audio.set('sfx', Number(e.target.value)));
  $('chkMute').addEventListener('change', e => { if (e.target.checked !== audio.prefs.muted) toggleMute(); });
  $('chkControls').addEventListener('change', e => { S.ctlOverride = e.target.checked; controlsVisibility(); });
  $('selQuality').addEventListener('change', e => { try { const u = new URL(root.location.href); u.searchParams.set('quality', e.target.value); toast('تُطبَّق الجودة الجديدة عند إعادة التحميل', 2); S.pendingQuality = u.toString(); } catch (x) { /* */ } });
  $('btnNext').addEventListener('click', nextStage);
  $('btnRetry').addEventListener('click', retryAfterGameOver);
  $('btnGoMenu').addEventListener('click', () => { hide('gameOver'); openMenu(); });
  $('btnAgain').addEventListener('click', () => { hide('final'); openMenu(); });
  $('btnPrint').addEventListener('click', () => root.print());
  $('btnRotateOk').addEventListener('click', () => { S.rotateOk = true; resize(); });

  // ------------------------------------------------------------------ boot
  async function boot() {
    try {
      setLoad(0.1, 'الخطوط والخامات'); if (doc.fonts && doc.fonts.ready) await Promise.race([doc.fonts.ready, new Promise(r => setTimeout(r, 2500))]);
      await root.QRTex.ready(); setLoad(0.45, 'بطل قطر');
      await view.initBoy(); setLoad(0.8, 'كورنيش الدوحة');
      loadStage(-1); S.showcase = true;
      setLoad(1, ''); hide('loading'); openMenu(); resize();
      if (S.platform.on) { post('ready'); $('btnStart').disabled = !S.platform.configured; }
      if (!QA) requestAnimationFrame(t => { last = t; frame(t); });
      S.booted = true;
    } catch (e) { console.error(e); fatal('حدث خطأ أثناء التحميل: ' + (e && e.message || e)); }
  }

  // ------------------------------------------------------------------ QA / test hooks (deterministic stepping)
  const ready = boot();
  root.QHGame = {
    state: S, ready, view, audio, camera, renderer,
    cfg: () => cfg, setConfig: c => { cfg = Cfg.validate(c); menuInfo(); return cfg; }, reloadConfig: () => { cfg = Cfg.load(); menuInfo(); audio.setEnabled(cfg.music); return cfg; },
    engine: () => S.engine, start, beginStage, openMenu, answer, togglePause, toggleMute, nextStage, retry: retryAfterGameOver,
    input: v => { S.qaInput = Object.assign({ left: false, right: false, jump: false }, v || {}); },
    step(seconds, dt = 1 / 60, render = false) { const n = Math.round(seconds / dt); for (let i = 0; i < n; i++) update(dt); if (render) view.render(camera); return S.phase; },
    render() { view.render(camera); },
    correctPos() { if (!S.q || !S.q.pres) return -1; return S.q.pres.options.findIndex(o => Cfg.isCorrect(S.q.q, o, cfg.salt)); },
    hud
  };
})(window);
