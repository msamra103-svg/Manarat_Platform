 neckA.y, neckA.z, headA.x, headA.y, headA.z, 0.058);
          for (const c of arms) collideCapsule(this.ghutra, c[0].x, c[0].y, c[0].z, c[1].x, c[1].y, c[1].z, c[2] + 0.012);
        }
      }
      // tassels
      for (const tz of this.tassels) {
        this.toModel(B.head, tz.anchor, V[4]); tz.chain.pin(0, V[4].x, V[4].y, V[4].z);
        for (let s = 0; s < steps; s++) { tz.chain.integrate(hs, g.x, g.y, g.z + wind * 0.4, 0.97); for (let it = 0; it < 3; it++) { tz.chain.solve(); this.collideTorso(tz.chain, 0.036); } }
        const p = tz.chain.p;
        tz.parts.forEach((m, i) => { const a = V[5].fromArray(p, i * 3), b = V[6].fromArray(p, i * 3 + 3); m.position.copy(a).add(b).multiplyScalar(0.5); const d = V[7].subVectors(b, a); const len = d.length() || 1e-4; m.scale.set(1, len, 1); m.quaternion.setFromUnitVectors(V[4].set(0, 1, 0), d.divideScalar(len)); });
      }
      this.syncClothMeshes();
    }
    // keep cloth outside the torso: radial profile measured in bind space, evaluated in the spine01 frame
    collideTorso(cl, margin) {
      const B = this.bones, sp = B.spine01, V = this._v;
      const toBone = this._toBone || (this._toBone = new T.Matrix4()); toBone.copy(sp.matrixWorld).invert().multiply(this.model.matrixWorld);
      const fromBone = this._fromBone || (this._fromBone = new T.Matrix4()); fromBone.copy(this._mi).multiply(sp.matrixWorld);
      const hd = sp.userData.head, nz = this.D.lm.neck[2], prof = this.torsoProf, y0 = prof[0].y, y1 = prof[prof.length - 1].y;
      const p = cl.p, v = V[0];
      for (let i = 0; i < cl.n; i++) {
        if (!cl.w[i]) continue;
        v.fromArray(p, i * 3).applyMatrix4(toBone).add(hd);       // → bind space
        if (v.y < y0 - 0.05 || v.y > y1) continue;
        const dx = v.x, dz = v.z - nz, rr = Math.hypot(dx, dz) || 1e-5;
        const th = Math.atan2(dx, dz), R = profileR(prof, clamp(v.y, y0, y1), th) + margin;
        if (rr < R) { v.x = dx / rr * R; v.z = nz + dz / rr * R; v.sub(hd).applyMatrix4(fromBone); p[i * 3] = v.x; p[i * 3 + 1] = v.y; p[i * 3 + 2] = v.z; }
      }
    }
    syncClothMeshes() {
      const NR = this.skirtNR, NC = this.skirtNC, pos = this.skirtMesh.geometry.attributes.position.array;
      // one Laplacian pass on the rendered surface (simulation untouched) — removes sub-grid crumpling
      const raw = this.skirt.p, sp = this._skS || (this._skS = new Float32Array(raw.length));
      for (let i = 0; i < NR; i++) for (let j = 0; j < NC; j++) {
        const k = (i * NC + j) * 3;
        if (i < 2 || i === NR - 1) { sp[k] = raw[k]; sp[k + 1] = raw[k + 1]; sp[k + 2] = raw[k + 2]; continue; }
        const a = ((i - 1) * NC + j) * 3, b = ((i + 1) * NC + j) * 3, c = (i * NC + (j + 1) % NC) * 3, d = (i * NC + (j + NC - 1) % NC) * 3;
        for (let q = 0; q < 3; q++) sp[k + q] = raw[k + q] * 0.5 + (raw[a + q] + raw[b + q] + raw[c + q] + raw[d + q]) * 0.125;
      }
      for (let i = 0; i < NR; i++) { for (let j = 0; j < NC; j++) { const s = (i * NC + j) * 3, d = (i * (NC + 1) + j) * 3; pos[d] = sp[s]; pos[d + 1] = sp[s + 1]; pos[d + 2] = sp[s + 2]; } const s = (i * NC) * 3, d = (i * (NC + 1) + NC) * 3; pos[d] = sp[s]; pos[d + 1] = sp[s + 1]; pos[d + 2] = sp[s + 2]; }
      this.skirtMesh.geometry.attributes.position.needsUpdate = true; this.skirtMesh.geometry.computeVertexNormals();
      const gp = this.ghutraMesh.geometry.attributes.position; gp.array.set(this.ghutra.p); gp.needsUpdate = true; this.ghutraMesh.geometry.computeVertexNormals();
    }
    settle(n = 240) { for (let i = 0; i < n; i++) this.simulate(1 / 120, 0); }

    // world-space helpers for gameplay/camera
    headWorld(out) { return this.bones.head.getWorldPosition(out); }
  }

  // ------------------------------------------------------------------ Animator: produces a pose (plain numbers, blendable)
  const KEYS = ['rx', 'ry', 'rz', 'rp', 'ryaw', 'rr', 'sp', 'sy', 'sr', 'np', 'ny', 'nr', 'shrug',
    'Lax', 'Lay', 'Laz', 'Lfp', 'Lko', 'Rax', 'Ray', 'Raz', 'Rfp', 'Rko',
    'Lsw', 'Lab', 'Lel', 'Ltw', 'Lhc', 'Lwf', 'Rsw', 'Rab', 'Rel', 'Rtw', 'Rhc', 'Rwf'];
  function blank() { const p = {}; for (const k of KEYS) p[k] = 0; return p; }
  function mix(out, a, b, t) { for (const k of KEYS) out[k] = a[k] + (b[k] - a[k]) * t; return out; }
  function catmull(pts, ts, u) { // pts: [[z,y],...], ts: knot times
    let k = 0; while (k < ts.length - 2 && u > ts[k + 1]) k++;
    const t = (u - ts[k]) / (ts[k + 1] - ts[k]);
    const p0 = pts[Math.max(0, k - 1)], p1 = pts[k], p2 = pts[k + 1], p3 = pts[Math.min(pts.length - 1, k + 2)];
    const t2 = t * t, t3 = t2 * t;
    const f = i => 0.5 * ((2 * p1[i]) + (-p0[i] + p2[i]) * t + (2 * p0[i] - 5 * p1[i] + 4 * p2[i] - p3[i]) * t2 + (-p0[i] + 3 * p1[i] - 3 * p2[i] + p3[i]) * t3);
    return [f(0), f(1)];
  }
  const lin = (x, xs, ys) => { if (x <= xs[0]) return ys[0]; for (let i = 1; i < xs.length; i++) if (x <= xs[i]) return lerp(ys[i - 1], ys[i], (x - xs[i - 1]) / (xs[i] - xs[i - 1])); return ys[ys.length - 1]; };

  class Animator {
    constructor(boy) {
      this.boy = boy; this.phase = 0; this.pose = blank(); this.prev = blank(); this.cur = blank();
      this.ankleH = boy.legs.L.restAnkle.y + SOLE; this.hipRestY = boy.rootRest.y + SOLE;
      this.state = 'idle'; this.stateT = 0; this.fade = 1; this.fadeDur = 0.2; this.from = blank();
      this.onFootstep = null; this.lastContact = { L: false, R: false };
      this.lean = 0; this.leanVel = 0; this.speedSm = 0;
    }
    set(state, fade = 0.18) {
      if (state === this.state) return;
      this.from = Object.assign({}, this.pose); this.state = state; this.stateT = 0; this.fade = 0; this.fadeDur = fade;
    }
    // ---- gait (walk ↔ run, forward or backward): foot-planted stance, spline swing
    gait(v, p, out, opt = {}) {
      const a = Math.abs(v), dirS = v < 0 ? -1 : 1;
      const R = smooth(1.9, 3.6, a);                     // 0 walk … 1 run
      const Tp = lin(a, [0, 0.8, 1.6, 3, 5, 7, 9, 12], [1.15, 1.05, 0.9, 0.72, 0.62, 0.55, 0.51, 0.46]);
      let ds = lin(a, [0, 0.8, 2, 2.6, 4, 7, 10, 12], [0.66, 0.64, 0.58, 0.42, 0.31, 0.2, 0.15, 0.13]);
      let sweep = a * Tp * ds; const maxS = lerp(0.6, 0.56, R); if (sweep > maxS) { ds = maxS / (a * Tp); sweep = maxS; }
      const lift = clamp(sweep / 0.5, 0.25, 1) * (opt.liftScale || 1);
      const zf = sweep * (R > 0.5 ? 0.36 : 0.48), zb = sweep - zf;
      const legs = {};
      for (const sd of ['L', 'R']) {
        const q = ((p + (sd === 'R' ? 0.5 : 0)) % 1 + 1) % 1, s = sd === 'L' ? 1 : -1;
        let z, y, fp, contact = false;
        if (q < ds) {
          const u = q / ds; z = zf - sweep * u; y = 0; contact = true;
          fp = R > 0.5 ? -0.05 + 0.6 * smooth(0.55, 1, u) : lerp(-0.18, 0, smooth(0, 0.25, u)) + 0.55 * smooth(0.6, 1, u);
          y += (R > 0.5 ? 0.05 : 0.07) * smooth(0.6, 1, u) ** 2;
        } else {
          const u = (q - ds) / (1 - ds);
          const pts = [[-zb, 0], [-zb - 0.03 - 0.03 * R, (0.05 + 0.07 * R) * lift], [-0.06 - 0.1 * R, (0.07 + 0.19 * R) * lift], [0.06 + 0.07 * R, (0.06 + 0.15 * R) * lift], [zf + 0.03 + 0.05 * R, (0.025 + 0.05 * R) * lift], [zf, 0]];
          [z, y] = catmull(pts, [0, 0.12, 0.36, 0.62, 0.86, 1], u);
          fp = lin(u, [0, 0.12, 0.45, 0.85, 1], [0.6, 0.7 * (0.5 + 0.5 * R), 0.25, R > 0.5 ? 0.0 : -0.2, R > 0.5 ? -0.05 : -0.18]);
        }
        if (dirS < 0) { z = -z; fp = fp * 0.5; }
        if (contact && !this.lastContact[sd] && this.onFootstep && a > 0.3) this.onFootstep(sd, a);
        this.lastContact[sd] = contact;
        legs[sd] = { x: s * lerp(0.092, 0.072, R), y: this.ankleH + Math.max(0, y), z: z + 0.015 + (opt.zBias || 0), fp };
      }
      const bobA = lerp(0.016, -0.026, R) * clamp(a / 2.5, 0.2, 1);
      const bob = bobA * Math.cos(TAU * 2 * (p - ds / 2));
      out.ry = -lerp(0.01, 0.034, R) * clamp(a / 3, 0, 1) + bob + (opt.drop || 0);
      out.rz = lerp(0.0, 0.03, R) * dirS; out.rx = 0;
      out.rp = lerp(0.02, 0.08, R) * dirS;
      out.ryaw = -lerp(0.07, 0.05, R) * Math.cos(TAU * p) * dirS;
      out.rr = lerp(0.03, 0.02, R) * Math.sin(TAU * p * 2) * 0.5;
      out.sp = (lerp(0.03, 0.14, R) + (opt.lean || 0)) * (dirS < 0 ? -0.6 : 1);
      out.sy = lerp(0.09, 0.12, R) * Math.cos(TAU * p) * dirS;
      out.sr = 0;
      out.np = -out.sp * 0.75 - out.rp * 0.9 - bob * 0.8;
      out.ny = -out.sy * 0.6 - out.ryaw * 0.7; out.nr = 0; out.shrug = 0.02 * R;
      for (const sd of ['L', 'R']) { out[sd + 'ax'] = legs[sd].x; out[sd + 'ay'] = legs[sd].y; out[sd + 'az'] = legs[sd].z; out[sd + 'fp'] = legs[sd].fp; out[sd + 'ko'] = 0.12; }
      const S = lerp(0.32, 0.72, R) * clamp(a / 3, 0.3, 1);
      const swL = -S * Math.cos(TAU * p) * dirS + 0.04 * R, swR = S * Math.cos(TAU * p) * dirS + 0.04 * R;
      for (const [sd, sw] of [['L', swL], ['R', swR]]) {
        out[sd + 'sw'] = sw; out[sd + 'ab'] = lerp(0.13, 0.2, R); out[sd + 'el'] = lerp(0.28, 1.35, R) + lerp(0.1, 0.35, R) * clamp(sw / Math.max(S, 0.01), 0, 1);
        out[sd + 'tw'] = lerp(0.0, 0.3, R) * clamp(0.5 + sw, 0, 1.2); out[sd + 'hc'] = lerp(0.3, 0.85, R); out[sd + 'wf'] = -0.1;
      }
      return Tp;
    }
    idle(t, out) {
      const br = Math.sin(t * 2.1);
      Object.assign(out, blank());
      out.ry = -0.008 + 0.003 * br; out.rx = 0.008 * Math.sin(t * 0.45); out.rr = -0.01 * Math.sin(t * 0.45);
      out.sp = 0.02 + 0.006 * br; out.np = -0.02; out.shrug = 0.01 * br;
      out.Lax = 0.1; out.Lay = this.ankleH; out.Laz = 0.02; out.Rax = -0.1; out.Ray = this.ankleH; out.Raz = -0.01; out.Lko = out.Rko = 0.1;
      for (const sd of ['L', 'R']) { out[sd + 'sw'] = 0.04; out[sd + 'ab'] = 0.11; out[sd + 'el'] = 0.24; out[sd + 'hc'] = 0.4; out[sd + 'tw'] = 0; out[sd + 'wf'] = -0.05; }
      return out;
    }
    victory(t, out) {
      this.idle(t, out);
      const hop = Math.max(0, Math.sin(t * 6.2)); out.ry += 0.035 * hop;
      for (const sd of ['L', 'R']) { out[sd + 'sw'] = 2.55 + 0.1 * Math.sin(t * 6.2); out[sd + 'ab'] = 0.42; out[sd + 'el'] = 0.35; out[sd + 'hc'] = 1.0; }
      out.np = -0.12; out.sp = -0.04;
      return out;
    }
    jump(j, out) { // j: 0..1 airborne progress
      Object.assign(out, blank());
      const tuck = Math.sin(Math.PI * clamp(j * 1.15, 0, 1));
      const land = smooth(0.62, 1, j);
      out.ry = -0.02 + 0.04 * tuck; out.sp = 0.16 + 0.1 * tuck - 0.12 * land; out.np = -out.sp * 0.7; out.rp = 0.05;
      // hurdle style: right leg leads, left leg tucked behind
      out.Rax = -0.08; out.Ray = this.ankleH + lerp(0.08, 0.3, tuck) * (1 - land) + 0.06 * land; out.Raz = lerp(0.12, 0.34, tuck) * (1 - land) + 0.06 * land; out.Rfp = lerp(0.4, -0.05, tuck);
      out.Lax = 0.085; out.Lay = this.ankleH + lerp(0.2, 0.44, tuck) * (1 - land) + 0.08 * land; out.Laz = lerp(-0.22, -0.18, tuck) * (1 - land) - 0.04 * land; out.Lfp = 0.75 - 0.5 * land;
      out.Lko = out.Rko = 0.12;
      out.Lsw = lerp(0.5, 1.05, tuck); out.Lel = 1.25; out.Lab = 0.22; out.Rsw = lerp(-0.2, -0.55, tuck); out.Rel = 0.95; out.Rab = 0.26;
      out.Lhc = out.Rhc = 0.75; out.Ltw = 0.2; out.Rtw = 0.1;
      return out;
    }
    landing(k, out, base) { // k: 0..1 absorb; base: a running pose to blend into
      const dip = Math.sin(Math.PI * clamp(k, 0, 1));
      Object.assign(out, base); out.ry -= 0.11 * dip; out.sp += 0.12 * dip; out.np -= 0.08 * dip;
      for (const sd of ['L', 'R']) { out[sd + 'ab'] += 0.12 * dip; }
      return out;
    }
    slide(k, out) { // k: 0..1 over the slide
      Object.assign(out, blank());
      const inn = smooth(0, 0.16, k), outW = 1 - smooth(0.82, 1, k), w = inn * outW;
      const drop = this.hipRestY - 0.33;
      out.ry = -drop * w; out.rz = -0.06 * w; out.rp = -0.42 * w; out.rr = 0.08 * w; out.ryaw = 0.12 * w;
      out.sp = -0.18 * w + 0.12 * (1 - w); out.sy = -0.1 * w; out.np = 0.62 * w; out.ny = -0.08 * w;
      // lead leg (L) extended forward along the ground, trail leg (R) folded underneath, knee outward
      out.Lax = lerp(0.08, 0.1, w); out.Lay = this.ankleH + lerp(0.02, 0.06, w); out.Laz = lerp(0.05, 0.64, w); out.Lfp = -0.35 * w; out.Lko = 0.15;
      out.Rax = lerp(-0.08, 0.02, w); out.Ray = this.ankleH + lerp(0.02, 0.02, w); out.Raz = lerp(-0.05, 0.2, w); out.Rfp = 0.6 * w; out.Rko = lerp(0.1, 2.2, w);
      out.Rsw = lerp(0.1, -0.55, w); out.Rab = lerp(0.15, 0.55, w); out.Rel = lerp(0.6, 0.15, w); out.Rhc = 0.3;
      out.Lsw = lerp(0.2, 0.85, w); out.Lab = lerp(0.15, 0.32, w); out.Lel = lerp(0.8, 0.55, w); out.Lhc = 0.6;
      return out;
    }
    recoil(t, out, scratch) { // t seconds since the wave hit; backward stumble then regain balance
      const back = 1 - smooth(0.0, 1.25, t);
      this.gait(-lerp(0.6, 2.3, back), this.phase, out, { lean: -0.32 * back, liftScale: 0.7 });
      const guard = smooth(0, 0.12, t) * (1 - smooth(0.55, 1.1, t));
      for (const sd of ['L', 'R']) { out[sd + 'sw'] = lerp(out[sd + 'sw'], 1.0, guard); out[sd + 'el'] = lerp(out[sd + 'el'], 1.7, guard); out[sd + 'ab'] = lerp(out[sd + 'ab'], 0.32, guard); out[sd + 'hc'] = lerp(out[sd + 'hc'], 0.5, guard); }
      out.np += -0.25 * guard; out.sp -= 0.12 * guard;
      return out;
    }
    // ---- main update: ctx = { state, speed, jumpT, slideT, landT, recoilT, lateral, t }
    update(dt, ctx) {
      const b = this.boy; if (ctx.state !== this.state) this.set(ctx.state, ctx.fade == null ? 0.18 : ctx.fade); this.stateT += dt;
      const cur = this.cur, v = ctx.speed;
      this.speedSm = lerp(this.speedSm, v, 1 - Math.exp(-dt * 8));
      switch (ctx.state) {
        case 'run': case 'walk': case 'decel': {
          const Tp = this.gait(this.speedSm, this.phase, cur, { lean: ctx.accelLean || 0 });
          this.phase = (this.phase + dt / Tp) % 1; break;
        }
        case 'jump': this.jump(ctx.jumpT, cur); this.phase = 0.1; break;
        case 'land': { const base = blank(); const Tp = this.gait(this.speedSm, this.phase, base, {}); this.phase = (this.phase + dt / Tp) % 1; this.landing(ctx.landT, cur, base); break; }
        case 'slide': this.slide(ctx.slideT, cur); break;
        case 'recoil': { this.recoil(ctx.recoilT, cur); const Tp = 0.62; this.phase = (this.phase + dt / Tp) % 1; break; }
        case 'victory': this.victory(ctx.t, cur); break;
        default: this.idle(ctx.t, cur);
      }
      // lane-change lean (critically damped) — roll into the turn, slight hip shift
      const target = clamp(-ctx.lateral * 0.065, -0.2, 0.2);
      this.leanVel += ((target - this.lean) * 120 - this.leanVel * 22) * dt; this.lean += this.leanVel * dt;
      cur.rr += this.lean; cur.sr += this.lean * 0.4; cur.nr -= this.lean * 0.9; cur.ny += clamp(ctx.lateral * 0.02, -0.2, 0.2);
      this.fade = Math.min(1, this.fade + dt / Math.max(this.fadeDur, 1e-3));
      const w = smooth(0, 1, this.fade);
      mix(this.pose, this.from, cur, w);
      b.applyPose(this.pose);
    }
  }

  root.QRBoy = { Boy, Animator, SOLE, blankPose: blank, KEYS };
})(typeof window !== 'undefined' ? window : globalThis);

<script data-src="js/hero-anim.js">/* مغامرات بطل قطر — platformer animation layer for the original Qatari boy (js/boy.js, unchanged).
 * Extends the runner's foot-planted Animator with the poses a platformer needs, all driven by real physics
 * values (never canned clips): variable-height airborne pose from vertical velocity (fist raised for the
 * head-bump while rising, arms out and legs reaching for the ground while falling), landing absorb,
 * skid/brake on direction change, stomp bounce tuck, hurt flinch, edge balance, look-up at a question block,
 * pit fall, victory.  Every state cross-fades (no pops); the character's meshes/materials are never touched.
 *
 * Model space: the boy faces +Z, his left is +X.  `near` = the side that faces the camera ('L' | 'R').
 */
(function (root) {
  'use strict';
  const B = root.QRBoy, T = root.THREE;
  const clamp = (x, a, b) => Math.max(a, Math.min(b, x));
  const lerp = (a, b, t) => a + (b - a) * t;
  const smooth = (a, b, x) => { const t = clamp((x - a) / (b - a), 0, 1); return t * t * (3 - 2 * t); };
  const blank = B.blankPose, KEYS = B.KEYS;
  const mix = (out, a, b, t) => { for (const k of KEYS) out[k] = a[k] + (b[k] - a[k]) * t; return out; };

  class HeroAnimator extends B.Animator {
    constructor(boy) {
      super(boy);
      this.base = blank(); this.tmp = blank(); this.lookUp = 0; this.lookUpW = 0; this.bob = 0;
      this.airPose = blank(); this.idleLook = 0; this.idleLookT = 3;
    }
    // airborne: u = vertical velocity normalised (+1 strong rise … −1 fast fall)
    air(vy, out, near, opt = {}) {
      Object.assign(out, blank());
      const up = smooth(-2.2, 3.2, vy), fall = 1 - up, tuck = opt.tuck || 0;
      const far = near === 'L' ? 'R' : 'L', sN = near === 'L' ? 1 : -1;
      // torso: forward lean on take-off, upright + slight arch at the apex, chest open while falling
      out.sp = 0.05 + 0.13 * up - 0.05 * fall + 0.18 * tuck; out.rp = 0.04 * up;
      out.np = -out.sp * 0.65 - 0.22 * up * (opt.reach == null ? 1 : opt.reach) + 0.05 * fall;
      out.ry = 0.015 * up - 0.03 * tuck; out.shrug = 0.03 * fall;
      // legs: far leg tucked forward-up (knee drive), near leg trails back; falling → both reach down/forward
      const lead = far, trail = near, ah = this.ankleH;
      out[lead + 'ax'] = -sN * 0.085; out[lead + 'ay'] = ah + lerp(0.1, 0.36, up) + 0.18 * tuck; out[lead + 'az'] = lerp(0.12, 0.24, up) + 0.06 * tuck; out[lead + 'fp'] = lerp(0.05, 0.55, up);
      out[trail + 'ax'] = sN * 0.09; out[trail + 'ay'] = ah + lerp(0.05, 0.24, up) + 0.22 * tuck; out[trail + 'az'] = lerp(-0.04, -0.24, up) + 0.12 * tuck; out[trail + 'fp'] = lerp(0.25, 0.85, up);
      out.Lko = out.Rko = 0.12;
      // arms: near arm punches up (head-bump fist), far arm swings back; falling → both out for balance
      const reach = opt.reach == null ? 1 : opt.reach;
      out[near + 'sw'] = lerp(0.55, 2.8, up * reach) - 0.3 * tuck; out[near + 'ab'] = lerp(0.95, 0.2, up); out[near + 'el'] = lerp(0.75, 0.22, up * reach); out[near + 'hc'] = lerp(0.45, 1.0, up); out[near + 'tw'] = 0.1; out[near + 'wf'] = -0.1;
      out[far + 'sw'] = lerp(0.35, -0.6, up) + 0.4 * tuck; out[far + 'ab'] = lerp(1.0, 0.3, up); out[far + 'el'] = lerp(0.7, 1.05, up); out[far + 'hc'] = 0.6; out[far + 'tw'] = 0.15; out[far + 'wf'] = -0.1;
      return out;
    }
    skid(k, out, near) { // braking on a direction change: heels dug in, weight back, arms forward
      Object.assign(out, blank());
      const w = Math.sin(Math.PI * clamp(k, 0, 1)) * 0.6 + 0.4, ah = this.ankleH;
      out.ry = -0.07 * w; out.rp = -0.12 * w; out.sp = -0.12 * w; out.np = 0.12 * w; out.rz = -0.04 * w;
      out.Lax = 0.1; out.Lay = ah + 0.01; out.Laz = 0.26 * w; out.Lfp = -0.35 * w; out.Lko = 0.14;
      out.Rax = -0.1; out.Ray = ah; out.Raz = -0.04; out.Rfp = 0.1; out.Rko = 0.14;
      for (const sd of ['L', 'R']) { out[sd + 'sw'] = 0.9 * w; out[sd + 'ab'] = 0.45 * w + 0.15; out[sd + 'el'] = 0.7; out[sd + 'hc'] = 0.5; out[sd + 'tw'] = 0.1; out[sd + 'wf'] = -0.2; }
      void near; return out;
    }
    balance(t, out) { // teetering at a ledge: windmilling arms, hips swaying
      this.idle(t, out);
      const s = Math.sin(t * 5.2), c = Math.cos(t * 5.2);
      out.rr = 0.07 * s; out.sr = -0.12 * s; out.nr = 0.08 * s; out.sp = 0.12 + 0.05 * c; out.np = -0.05;
      out.Lsw = 1.2 + 0.9 * s; out.Rsw = 1.2 - 0.9 * s; out.Lab = out.Rab = 1.15; out.Lel = out.Rel = 0.35; out.Lhc = out.Rhc = 0.3;
      out.Laz = 0.06; out.Raz = -0.02; out.Lay = this.ankleH + Math.max(0, 0.03 * s); out.Ray = this.ankleH + Math.max(0, -0.03 * s);
      return out;
    }
    hurt(t, out, near) { // flinch from a hit (knock-back moves the body backwards, the gait follows)
      this.recoil(clamp(t * 1.3, 0, 2), out);
      const k = 1 - smooth(0.15, 0.7, t);
      out.sp -= 0.2 * k; out.np += 0.18 * k; out.ry -= 0.05 * k;
      out[near + 'sw'] = lerp(out[near + 'sw'], 1.4, k); out[near + 'el'] = lerp(out[near + 'el'], 1.6, k);
      return out;
    }
    pitfall(t, out, near) { // falling into the water: both arms up, legs kicking
      this.air(-8, out, near, { reach: 0 });
      const s = Math.sin(t * 13);
      out.Lsw = out.Rsw = 2.5 + 0.25 * s; out.Lab = out.Rab = 0.55; out.Lel = out.Rel = 0.3; out.np = -0.3;
      out.Lay += 0.08 * Math.max(0, s); out.Ray += 0.08 * Math.max(0, -s);
      return out;
    }
    idleLife(t, out) { // breathing + occasional look around (towards the camera) when standing still
      this.idle(t, out);
      out.ny += 0.35 * this.idleLook; out.np -= 0.04 * Math.abs(this.idleLook);
      return out;
    }
    /* ctx = { state: idle|run|skid|air|land|hurt|balance|victory|pitfall, speed (|vx|), vy, near, t, landT, landK,
     *         skidT, hurtT, tuck, lookUp (0..1, question block above), accel } */
    update(dt, ctx) {
      const b = this.boy; if (ctx.state !== this.state) this.set(ctx.state, ctx.fade == null ? 0.16 : ctx.fade); this.stateT += dt;
      const cur = this.cur, base = this.base, near = ctx.near || 'R';
      this.speedSm = lerp(this.speedSm, ctx.speed, 1 - Math.exp(-dt * 10));
      // idle head look-around schedule
      this.idleLookT -= dt; if (this.idleLookT < 0) { this.idleLookT = 2.5 + Math.random() * 3; this.idleLookTarget = this.idleLookTarget ? 0 : (Math.random() < 0.6 ? 0.9 : -0.5); }
      this.idleLook = lerp(this.idleLook, ctx.state === 'idle' && ctx.lookUp < 0.1 ? (this.idleLookTarget || 0) : 0, 1 - Math.exp(-dt * 3));
      switch (ctx.state) {
        case 'run': {
          if (this.speedSm < 0.12) { this.idleLife(ctx.t, cur); break; }
          const Tp = this.gait(this.speedSm, this.phase, cur, { lean: clamp(ctx.accel || 0, -1, 1) * 0.08 });
          this.phase = (this.phase + dt / Tp) % 1;
          // starting from a stand: fade from the idle pose while the speed builds (no foot pop)
          const w = smooth(0.12, 0.9, this.speedSm); if (w < 1) { this.idleLife(ctx.t, base); mix(cur, base, cur, w); }
          break;
        }
        case 'skid': this.skid(ctx.skidT || 0, cur, near); break;
        case 'air': this.air(ctx.vy, cur, near, { tuck: ctx.tuck || 0 }); this.phase = near === 'L' ? 0.05 : 0.55; if (this.stateT < 0.02) this.fadeDur = 0.09; break;
        case 'land': {
          const k = clamp(ctx.landT / 0.22, 0, 1);
          if (this.speedSm > 0.25) { const Tp = this.gait(this.speedSm, this.phase, base, {}); this.phase = (this.phase + dt / Tp) % 1; } else this.idleLife(ctx.t, base);
          this.landing(k, cur, base); const d = Math.sin(Math.PI * k) * (ctx.landK || 1);
          cur.ry -= 0.03 * d; for (const sd of ['L', 'R']) cur[sd + 'el'] += 0.25 * d;
          break;
        }
        case 'hurt': { this.hurt(ctx.hurtT || 0, cur, near); this.phase = (this.phase + dt / 0.6) % 1; break; }
        case 'balance': this.balance(ctx.t, cur); break;
        case 'victory': this.victory(ctx.t, cur); break;
        case 'pitfall': this.pitfall(ctx.t, cur, near); break;
        default: this.idleLife(ctx.t, cur);
      }
      // look up at a question block overhead (guidance cue): head and eyes tilt up, chest opens
      this.lookUpW = lerp(this.lookUpW, clamp(ctx.lookUp || 0, 0, 1), 1 - Math.exp(-dt * 6));
      if (this.lookUpW > 0.001 && ctx.state !== 'air') { cur.np -= 0.55 * this.lookUpW; cur.sp -= 0.06 * this.lookUpW; }
      this.fade = Math.min(1, this.fade + dt / Math.max(this.fadeDur, 1e-3));
      mix(this.pose, this.from, cur, smooth(0, 1, this.fade));
      b.applyPose(this.pose);
    }
  }
  root.QHAnim = { HeroAnimator };

<script data-src="js/levels.js">/* مغامرات بطل قطر — stage layouts.
 * A stage is built by a small builder into: a solid tile grid (1 m tiles), interactive blocks, coins, enemies,
 * moving / falling platforms, checkpoints, the goal and up to six QUESTION SLOTS.  Every slot is a lone
 * question block 3 m above flat ground inside a guaranteed safe zone (no pit, no enemy, no other block in
 * the jump column), so it can always be reached and hit — the automated tests replay every slot.
 *
 * Coordinates: x grows to the right (the hero's travel direction), y up; ground surface at y = 0.
 * Grid cell (ix, iy) covers x ∈ [ix, ix+1], y ∈ [iy, iy+1]; iy runs from Y_MIN (−4) to Y_MAX.
 */
(function (root) {
  'use strict';
  const Y_MIN = -4, Y_MAX = 14, H = Y_MAX - Y_MIN;
  // solid tile kinds (render + rules)
  const K = { EMPTY: 0, GROUND: 1, TERRACE: 2, CRATE: 3, LEDGE: 4, BRICK: 5, QBLOCK: 6, COINBLOCK: 7, ITEMBLOCK: 8, USED: 9, STONE: 10, HIDDEN: 11 };
  const INTERACTIVE = new Set([K.BRICK, K.QBLOCK, K.COINBLOCK, K.ITEMBLOCK, K.USED, K.HIDDEN]);

  class Builder {
    constructor(id, width) {
      this.id = id; this.W = width; this.grid = new Uint8Array(width * H);
      this.blockMap = new Map(); this.coins = []; this.enemies = []; this.movers = []; this.signs = []; this.deco = [];
      this.slots = []; this.checkpoints = []; this.safe = []; this.start = { x: 3, y: 0 }; this.goal = null; this.notes = [];
    }
    idx(ix, iy) { return ix + (iy - Y_MIN) * this.W; }
    set(ix, iy, k) { if (ix < 0 || ix >= this.W || iy < Y_MIN || iy >= Y_MAX) return; this.grid[this.idx(ix, iy)] = k; }
    get(ix, iy) { if (ix < 0 || ix >= this.W || iy < Y_MIN || iy >= Y_MAX) return 0; return this.grid[this.idx(ix, iy)]; }
    // ground columns [x0, x1) with surface at `top` (fills down to the bottom of the world)
    ground(x0, x1, top = 0, kind = K.GROUND) { for (let x = x0; x < x1; x++) for (let y = Y_MIN; y < top; y++) this.set(x, y, y < 0 || kind === K.GROUND ? K.GROUND : kind); return this; }
    terrace(x0, x1, top) { for (let x = x0; x < x1; x++) for (let y = Y_MIN; y < top; y++) this.set(x, y, y < 0 ? K.GROUND : K.TERRACE); return this; }
    pit(x0, x1) { for (let x = x0; x < x1; x++) for (let y = Y_MIN; y < Y_MAX; y++) this.set(x, y, K.EMPTY); return this; }
    crates(x, y, w = 1, h = 1) { for (let i = 0; i < w; i++) for (let j = 0; j < h; j++) this.set(x + i, y + j, K.CRATE); return this; }
    stone(x, y, w = 1, h = 1) { for (let i = 0; i < w; i++) for (let j = 0; j < h; j++) this.set(x + i, y + j, K.STONE); return this; }
    ledge(x, y, w) { for (let i = 0; i < w; i++) this.set(x + i, y, K.LEDGE); return this; }
    // a row of blocks: B brick · $ coin block (5 coins) · c coin block (1 coin) · + heart · 1 extra life · h hidden 1-up · . gap
    blocks(x, y, pattern) {
      [...pattern].forEach((ch, i) => {
        const bx = x + i; if (ch === '.') return;
        const kind = { B: K.BRICK, $: K.COINBLOCK, c: K.COINBLOCK, '+': K.ITEMBLOCK, '1': K.ITEMBLOCK, h: K.HIDDEN }[ch];
        if (kind == null) throw new Error('unknown block ' + ch);
        this.set(bx, y, kind);
        this.blockMap.set(bx + ',' + y, { x: bx, y, kind, coins: ch === '$' ? 6 : ch === 'c' ? 1 : 0, item: ch === '+' ? 'heart' : (ch === '1' || ch === 'h') ? 'life' : null });
      });
      return this;
    }
    coinRow(x, y, n, dx = 1) { for (let i = 0; i < n; i++) this.coins.push({ x: x + i * dx + 0.5, y: y + 0.5 }); return this; }
    coinArc(x0, x1, y0, h) { const n = Math.max(3, Math.round(x1 - x0)); for (let i = 0; i <= n; i++) { const t = i / n; this.coins.push({ x: x0 + (x1 - x0) * t, y: y0 + 4 * h * t * (1 - t) + 0.5 }); } return this; }
    enemy(type, x, y = 0, o = {}) { this.enemies.push(Object.assign({ type, x: x + 0.5, y, dir: -1 }, o)); return this; }
    // moving platform: kind boat (on water, horizontal) | lift (vertical) | float (horizontal in the air) | fall
    mover(kind, x, y, w, o = {}) { this.movers.push(Object.assign({ kind, x, y, w, axis: kind === 'lift' ? 'y' : 'x', range: 0, speed: 1, phase: 0 }, o)); return this; }
    // question slot: lone block 3 m above the ground at x; safe zone [x − 7, x + 6]
    slot(x, top = 0) {
      const y = top + 3;
      this.slots.push({ x, y, top, index: this.slots.length });
      this.safe.push([x - 7, x + 6]);
      return this;
    }
    checkpoint(x, top = 0) { this.checkpoints.push({ x: x + 0.5, y: top }); return this; }
    sign(x, text, top = 0, o = {}) { this.signs.push(Object.assign({ x: x + 0.5, y: top, text }, o)); return this; }
    decoAt(type, x, o = {}) { this.deco.push(Object.assign({ type, x }, o)); return this; }
    surfaceAt(ix) { for (let y = Y_MAX - 1; y >= Y_MIN; y--) { const k = this.get(ix, y); if (k === K.GROUND || k === K.TERRACE) return y + 1; } return null; }
    // ---- validation of the hard rules (thrown at build time and asserted by the tests)
    validate() {
      const errs = [];
      for (const s of this.slots) {
        for (let x = s.x - 7; x <= s.x + 6; x++) { const top = this.surfaceAt(x); if (top !== s.top) errs.push(`slot ${s.index}: ground not flat at x=${x} (top ${top} ≠ ${s.top})`); }
        for (let y = s.top; y < s.y; y++) for (let x = s.x - 1; x <= s.x + 1; x++) if (this.get(x, y)) errs.push(`slot ${s.index}: obstruction under block at ${x},${y}`);
        for (let x = s.x - 1; x <= s.x + 1; x++) if (x !== s.x && this.get(x, s.y) && INTERACTIVE.has(this.get(x, s.y))) errs.push(`slot ${s.index}: neighbour block at ${x}`);
        for (const e of this.enemies) if (e.x + (e.range || 0) > s.x - 9 && e.x - (e.range || 0) < s.x + 8) errs.push(`slot ${s.index}: enemy ${e.type} inside safe zone`);
        for (const m of this.movers) if (m.x + m.w + (m.axis === 'x' ? m.range : 0) > s.x - 7 && m.x < s.x + 6) errs.push(`slot ${s.index}: moving platform inside safe zone`);
      }
      for (let i = 1; i < this.slots.length; i++) if (this.slots[i].x - this.slots[i - 1].x < 16) errs.push('slots too close');
      if (this.surfaceAt(Math.floor(this.start.x)) == null) errs.push('no ground at start');
      if (!this.goal || this.surfaceAt(Math.floor(this.goal.x)) == null) errs.push('no ground at goal');
      if (errs.length) throw new Error(this.id + ' layout invalid:\n' + errs.join('\n'));
      return true;
    }
    finish(meta) {
      // slot blocks are real tiles in the grid
      for (const s of this.slots) { this.set(s.x, s.y, K.QBLOCK); this.blockMap.set(s.x + ',' + s.y, { x: s.x, y: s.y, kind: K.QBLOCK, slot: s.index, coins: 0, item: null }); }
      this.validate();
      return Object.assign({ id: this.id, W: this.W, H, Y_MIN, Y_MAX, grid: this.grid, blocks: this.blockMap, coins: this.coins, enemies: this.enemies, movers: this.movers,
        signs: this.signs, deco: this.deco, slots: this.slots, checkpoints: this.checkpoints, safe: this.safe, start: this.start, goal: this.goal }, meta);
    }
  }

<script data-src="js/engine.js">/* مغامرات بطل قطر — gameplay engine (render-agnostic, deterministic, fixed 120 Hz step).
 * Player controller (eased run, skid, variable-height jump with coyote time and input buffer, optional
 * teacher-enabled double jump, head-bump), tile + platform collision, interactive blocks, coins, items,
 * enemies with patrols, moving / falling platforms, checkpoints, goal — and the QUESTION GATES:
 *
 *   Every question slot that carries a question owns an invisible progress gate (a logical limit on x, at
 *   every height) that exists while the slot is unsolved.  Nothing can carry the hero past it: running,
 *   jumping, double jumping, platforms, knock-back and corner correction are all clamped in one place
 *   (`clampGate`).  The block can only be solved by hitting it from below, which raises a `question` event
 *   and freezes the stage until the game calls `solveSlot()` after a CORRECT answer.
 *   With auto-run on, the hero brakes to a stop under the next unsolved block and jumps to hit it.
 *
 * Events go to `onEvent(type, data)`; the view and audio listen, the engine never touches the DOM or WebGL.
 */
(function (root) {
  'use strict';
  const L = root.QHLevels, K = L.K;
  const clamp = (x, a, b) => Math.max(a, Math.min(b, x));
  const sign = x => x > 0 ? 1 : x < 0 ? -1 : 0;
  const STEP = 1 / 120;
  const P = {                       // tuning (metres, seconds); the hero is 1.47 m tall
    w: 0.56, h: 1.42, maxRun: 5.4, accG: 24, accA: 15, decG: 30, skidDec: 42, jumpV: 11.4, gUp: 25.5, gCut: 75, gDown: 36, terminal: -17,
    coyote: 0.1, buffer: 0.14, holdMax: 0.34, autoRun: 5.4, doubleV: 9.6, stompV: 9.0, stompHeldV: 12.2, hurtV: 6.5, hurtVX: 5.0, invul: 1.7, hurtCtl: 0.42
  };
  const SOLID = new Set([K.GROUND, K.TERRACE, K.CRATE, K.BRICK, K.QBLOCK, K.COINBLOCK, K.ITEMBLOCK, K.USED, K.STONE]);
  const ENEMY = {
    crab: { w: 0.78, h: 0.5, speed: 1.15, stomp: true, gravity: true, score: 100 },
    urchin: { w: 0.72, h: 0.62, speed: 0.55, stomp: false, gravity: true, score: 0 },
    gull: { w: 0.9, h: 0.45, speed: 1.6, stomp: true, gravity: false, score: 150 },
    // later stages reuse the same behaviours with their own looks (see view.js)
    lantern: { w: 0.62, h: 0.8, speed: 1.0, stomp: true, gravity: true, score: 100 },
    pigeon: { w: 0.9, h: 0.45, speed: 1.5, stomp: true, gravity: false, score: 150 },
    hedgehog: { w: 0.8, h: 0.6, speed: 0.6, stomp: false, gravity: true, score: 0 },
    robot: { w: 0.74, h: 0.55, speed: 1.2, stomp: true, gravity: true, score: 100 },
    spark: { w: 0.6, h: 0.6, speed: 0.8, stomp: false, gravity: true, score: 0 },
    locust: { w: 0.8, h: 0.4, speed: 1.7, stomp: true, gravity: false, score: 150 },
    drone: { w: 0.9, h: 0.4, speed: 1.7, stomp: true, gravity: false, score: 150 },
    mask: { w: 0.7, h: 0.75, speed: 1.1, stomp: true, gravity: true, score: 120, hop: true },
    scarab: { w: 0.74, h: 0.46, speed: 1.3, stomp: true, gravity: true, score: 100 },
    dust: { w: 0.8, h: 1.1, speed: 0.9, stomp: false, gravity: true, score: 0 },
    cactus: { w: 0.6, h: 1.0, speed: 0, stomp: false, gravity: true, score: 0 }
  };

  class Engine {
    constructor(level, opts = {}) {
      this.L = level; this.opts = Object.assign({ doubleJump: false, autorun: false, hearts: 3, difficulty: 'normal' }, opts);
      this.onEvent = opts.onEvent || (() => {});
      this.grid = level.grid.slice(); this.W = level.W;
      this.time = 0; this.acc = 0; this.frozen = false; this.stats = { coins: 0, score: 0, stomps: 0, bricks: 0 };
      // blocks
      this.blocks = new Map(); for (const [k, b] of level.blocks) this.blocks.set(k, Object.assign({ bumpT: -1, used: false, hidden: b.kind === K.HIDDEN }, b));
      // question slots: which carry a question (assigned by the game), solved state, gate limit
      this.slots = level.slots.map(s => ({ x: s.x, y: s.y, top: s.top, index: s.index, active: false, solved: false, qIndex: -1, gateX: s.x + 1.3, tries: 0 }));
      this.coins = level.coins.map((c, i) => ({ x: c.x, y: c.y, id: i, taken: false }));
      this.items = []; this.debris = 0;
      this.enemyDefs = level.enemies; this.resetEnemies();
      this.movers = level.movers.map((m, i) => this.makeMover(m, i));
      this.checkpoints = level.checkpoints.map(c => ({ x: c.x, y: c.y, on: false }));
      this.goal = level.goal; this.goalReached = false;
      this.respawn = { x: level.start.x, y: level.start.y };
      this.player = this.newPlayer(level.start.x, level.start.y);
      this.input = { left: false, right: false, jump: false };
      this.gatePushT = 0; this.nearSlot = null; this.auto = { state: 'run', jumpT: 0, waitT: 0 };
      this.chain = 0;
    }
    newPlayer(x, y) { return { x, y, vx: 0, vy: 0, w: P.w, h: P.h, onGround: true, ground: null, facing: 1, coyote: P.coyote, jumpBuf: 0, held: false, holdT: 0, airJumps: 0, jumping: false, invulT: 0, hurtT: 0, skidT: 0, landT: 9, landK: 0, fallStartY: y, hearts: this.opts.hearts, dead: false, prevBottom: y, airT: 0, bumpT: -1, tuck: 0, accel: 0, blockedT: 0 }; }
    // ---------------------------------------------------------------- setup helpers
    assignQuestions(map) { // map: slotIndex → question index (or −1 for no question)
      for (const s of this.slots) { const q = map[s.index]; s.active = q != null && q >= 0; s.qIndex = s.active ? q : -1; s.solved = false; s.tries = 0;
        const b = this.blockAt(s.x, s.y); if (!s.active && b) { b.kind = K.COINBLOCK; b.coins = 1; this.setTile(s.x, s.y, K.COINBLOCK); } }
    }
    resetEnemies() {
      this.enemies = this.enemyDefs.map((e, i) => { const d = ENEMY[e.type] || ENEMY.crab; return { id: i, type: e.type, def: d, x: e.x, y: e.y, x0: e.x, y0: e.y, vx: 0, vy: 0, dir: e.dir || -1, range: e.range || 0, speed: e.speed || d.speed, w: d.w, h: d.h, alive: true, deadT: -1, onGround: true, phase: (i * 1.7) % 6.28, hopT: 0, fly: !d.gravity }; });
    }
    makeMover(m, i) { return { id: i, kind: m.kind, x0: m.x, y0: m.y, x: m.x, y: m.y, w: m.w, h: m.kind === 'boat' ? 0.5 : 0.3, axis: m.axis, range: m.range, speed: m.speed, phase: m.phase || 0, t: 0, dx: 0, dy: 0, state: 'idle', fallT: 0, vy: 0, respawnT: 0 }; }
    tile(ix, iy) { if (ix < 0 || ix >= this.W || iy < L.Y_MIN || iy >= L.Y_MAX) return 0; return this.grid[ix + (iy - L.Y_MIN) * this.W]; }
    setTile(ix, iy, k) { if (ix < 0 || ix >= this.W || iy < L.Y_MIN || iy >= L.Y_MAX) return; this.grid[ix + (iy - L.Y_MIN) * this.W] = k; }
    blockAt(ix, iy) { return this.blocks.get(ix + ',' + iy) || null; }
    solid(ix, iy) { const k = this.tile(ix, iy); return SOLID.has(k); }
    oneWay(ix, iy) { return this.tile(ix, iy) === K.LEDGE; }
    // active gate = first unsolved active slot (they are ordered left → right; later gates are behind earlier ones)
    activeGate() { for (const s of this.slots) if (s.active && !s.solved) return s; return null; }
    clampGate(p) {
      const g = this.activeGate(); if (!g) return false;
      const lim = g.gateX - p.w / 2;
      if (p.x > lim) { p.x = lim; if (p.vx > 0) p.vx = 0; return true; }
      return false;
    }
    // ---------------------------------------------------------------- main step
    update(dt) {
      this.acc += Math.min(dt, 0.1);
      while (this.acc >= STEP - 1e-9) { this.acc -= STEP; this.step(STEP); }   // epsilon: a 120 Hz frame always runs exactly one step (no 0/2-step jitter that could swallow a 1-frame key release)
    }
    step(h) {
      if (this.frozen) return;
      this.time += h;
      for (const m of this.movers) this.stepMover(m, h);
      this.stepPlayer(h);
      for (const e of this.enemies) this.stepEnemy(e, h);
      for (const it of this.items) this.stepItem(it, h);
      this.items = this.items.filter(i => !i.gone);
      for (const b of this.blocks.values()) if (b.bumpT >= 0) { b.bumpT += h; if (b.bumpT > 0.26) b.bumpT = -1; }
      this.interact();
    }
    // ---------------------------------------------------------------- player
    // width of the unsupported gap starting ahead of x (columns with nothing to stand on near height y)
    gapAhead(x, y) {
      const sup = ix => { for (let iy = Math.floor(y + 0.5) - 1; iy >= Math.max(L.Y_MIN, Math.floor(y) - 4); iy--) if (this.solid(ix, iy) || this.oneWay(ix, iy)) return true; return false; };
      const level = ix => this.solid(ix, Math.floor(y - 0.5)) || this.oneWay(ix, Math.floor(y - 0.5));
      let ix = Math.floor(x + 0.05), start = null, w = 0, drop = null;
      for (let k = 0; k < 10; k++, ix++) { if (drop == null && !level(ix)) drop = ix; if (!sup(ix)) { if (start == null) start = ix; w++; } else if (start != null) break; }
      return { start, w, drop };
    }
    computeAutoInput(p) {
      // Classroom touch controls override auto-run. Holding stops instantly;
      // touching the left side (or pressing Left) reverses the hero.
      if (this.input.stop) return { left: false, right: false, jump: false };
      if (this.input.back || this.input.left) return { left: true, right: false, jump: false };
      const inp = { left: false, right: p.vx < P.autoRun, jump: this.input.jump };
      // standing on a boat / lift: stay on it (the student jumps off when the far side is close)
      if (p.onGround && p.ground && p.ground.kind !== 'fall') { const m = p.ground, c = m.x + m.w / 2; inp.right = p.x < c - 0.35; inp.left = p.x > c + 0.35; return inp; }
      // a gap too wide to jump (needs a platform): brake and wait at the edge
      if (p.onGround && !p.ground) { const g = this.gapAhead(p.x + p.w / 2, p.y); if (g.start != null && g.w > 3.5) { const edge = Math.min(g.start, g.drop == null ? 1e9 : g.drop) - p.w / 2 - 0.12, dist = edge - p.x; const want = Math.sqrt(2 * P.decG * 0.8 * Math.max(0, dist)); inp.right = dist > 0.02 && p.vx < want; inp.left = dist < -0.05; } }
      const g = this.activeGate();
      if (g && !this.frozen) {
        const cx = g.x + 0.5, dx = cx - p.x;
        if (dx < 9 && dx > -1.2) {
          // brake to stop exactly under the block, then jump with a full press to hit it
          const a = this.auto;
          if (p.onGround && Math.abs(dx) < 0.12 && Math.abs(p.vx) < 0.4) { a.waitT += STEP; inp.right = false; if (a.waitT > 0.25) { inp.jump = true; a.jumpT = 0.3; a.waitT = -0.9; } }
          else if (a.jumpT > 0) { inp.right = false; inp.jump = true; }
          else { const want = clamp(Math.sqrt(2 * P.decG * 0.9 * Math.max(0, dx)), 0, P.maxRun * 0.8); inp.right = dx > 0.05 && p.vx < want; inp.left = dx < -0.05; if (!p.onGround) { inp.right = false; } this.autoBrake = want; }
          if (a.jumpT > 0) a.jumpT -= STEP;
        }
      }
      return inp;
    }
    // safety net: never let the hero overlap solid tiles (bad spawn, platform squeeze) — push him up onto the surface
    unstick(p) {
      const [x0, x1] = this.overlapX(p), y0 = Math.floor(p.y + 0.05), y1 = Math.floor(p.y + p.h - 0.05);
      let hit = false; for (let ix = x0; ix <= x1 && !hit; ix++) for (let iy = y0; iy <= y1; iy++) if (this.solid(ix, iy)) { hit = true; break; }
      if (!hit) return false;
      for (let ny = y0 + 1; ny < L.Y_MAX; ny++) { let free = true; for (let ix = x0; ix <= x1 && free; ix++) for (let iy = ny; iy <= Math.floor(ny + p.h - 0.05); iy++) if (this.solid(ix, iy)) { free = false; break; } if (free) { p.y = ny; p.vy = 0; p.onGround = true; p.prevBottom = ny; this.onEvent('unstick', { x: p.x, y: ny }); return true; } }
      return false;
    }
    stepPlayer(h) {
      const p = this.player; if (p.dead) return;
      this.unstick(p);
      if (this.opts.autorun && this.input.stop) p.vx = 0;
      const inp = this.opts.autorun ? this.computeAutoInput(p) : this.input;
      p.prevBottom = p.y; p.invulT = Math.max(0, p.invulT - h); p.landT += h; p.airT = p.onGround ? 0 : p.airT + h;
      if (p.bumpT >= 0) { p.bumpT += h; if (p.bumpT > 0.3) p.bumpT = -1; }
      // ride the platform we stand on
      if (p.onGround && p.ground && p.ground.kind) { p.x += p.ground.dx; p.y += p.ground.dy; }
      // horizontal control
      const dir = (inp.right ? 1 : 0) - (inp.left ? 1 : 0), ctl = p.hurtT <= 0;
      if (p.hurtT > 0) p.hurtT -= h;
      const vPrev = p.vx;
      if (ctl && dir) {
        const opposing = sign(p.vx) && sign(p.vx) !== dir;
        const a = p.onGround ? (opposing ? P.skidDec : P.accG) : P.accA;
        p.vx += dir * a * h; if (Math.abs(p.vx) > P.maxRun) p.vx = sign(p.vx) * Math.max(P.maxRun, Math.abs(p.vx) - P.decG * h);
        if (p.onGround && opposing && Math.abs(p.vx) > 1.6) p.skidT += h; else p.skidT = 0;
        if (!opposing || Math.abs(p.vx) < 0.5 || !p.onGround) p.facing = dir;
      } else if (p.onGround && ctl) {
        const d = P.decG * h; p.vx = Math.abs(p.vx) <= d ? 0 : p.vx - sign(p.vx) * d; p.skidT = 0;
      } else if (!p.onGround && ctl) { const d = 4 * h; p.vx = Math.abs(p.vx) <= d ? 0 : p.vx - sign(p.vx) * d; }
      p.accel = (p.vx - vPrev) / h / P.accG;
      // jump: buffer + coyote time; hold for height; optional double jump
      if (inp.jump && !p.held) p.jumpBuf = P.buffer;
      if (p.jumpBuf > 0) p.jumpBuf -= h;
      if (p.onGround) { p.coyote = P.coyote; p.airJumps = this.opts.doubleJump ? 1 : 0; } else p.coyote -= h;
      if (p.jumpBuf > 0 && ctl) {
        if (p.coyote > 0) { if (p.ground && p.ground.kind) { p.vx += p.ground.dx / STEP; p.vy += Math.max(0, p.ground.dy / STEP); } p.vy = Math.max(p.vy, 0) + P.jumpV + Math.abs(p.vx) * 0.12; p.onGround = false; p.ground = null; p.coyote = 0; p.jumpBuf = 0; p.jumping = true; p.holdT = 0; p.fallStartY = p.y; this.onEvent('jump', { x: p.x, y: p.y, double: false }); }
        else if (p.airJumps > 0) { p.airJumps--; p.vy = P.doubleV; p.jumpBuf = 0; p.jumping = true; p.holdT = 0; p.tuck = 1; this.onEvent('jump', { x: p.x, y: p.y, double: true }); }
      }
      p.held = !!inp.jump;
      if (p.jumping && p.held && p.vy > 0) p.holdT += h; else p.jumping = p.jumping && p.vy > 0 && p.held;
      const g = p.vy > 0 ? ((p.jumping && p.held && p.holdT < P.holdMax) ? P.gUp : P.gCut) : P.gDown;
      p.vy = Math.max(P.terminal, p.vy - g * h);
      p.tuck = Math.max(0, p.tuck - h * 2.5);
      // move X then Y with tile resolution
      p.x += p.vx * h;
      this.resolveX(p);
      if (this.clampGate(p)) { this.gatePushT += h; } else this.gatePushT = Math.max(0, this.gatePushT - h * 2);
      p.x = clamp(p.x, p.w / 2 + 0.05, this.W - p.w / 2 - 0.05);
      const wasGround = p.onGround;
      p.y += p.vy * h;
      this.resolveY(p, wasGround);
      if (!wasGround && p.onGround) { const drop = p.fallStartY - p.y; p.landT = 0; p.landK = clamp(drop / 2.5, 0.25, 1); this.onEvent('land', { x: p.x, y: p.y, k: p.landK }); this.chain = 0; }
      if (p.onGround) p.fallStartY = p.y; else if (p.vy > 0) p.fallStartY = Math.max(p.fallStartY, p.y);
      // fell into the water / a pit
      if (p.y < L.Y_MIN + 0.5) { p.dead = true; this.onEvent('die', { cause: 'pit', x: p.x }); }
      // checkpoints & goal
      for (const c of this.checkpoints) if (!c.on && Math.abs(p.x - c.x) < 0.7 && p.y < c.y + 2.5) { c.on = true; this.respawn = { x: c.x, y: c.y }; this.onEvent('checkpoint', c); }
      if (!this.goalReached && this.goal && p.x >= this.goal.x - 0.45 && !this.activeGate()) { this.goalReached = true; this.onEvent('goal', { x: this.goal.x }); }
      // guidance: nearest unsolved active block ahead
      const gs = this.activeGate(); this.nearSlot = gs ? { index: gs.index, dist: gs.x + 0.5 - p.x, under: Math.abs(gs.x + 0.5 - p.x) < 0.9 && p.y < gs.y } : null;
    }
    overlapX(p) { return [Math.floor(p.x - p.w / 2 + 1e-4), Math.floor(p.x + p.w / 2 - 1e-4)]; }
    resolveX(p) {
      const y0 = Math.floor(p.y + 0.02), y1 = Math.floor(p.y + p.h - 0.02);
      if (p.vx > 0) { const ix = Math.floor(p.x + p.w / 2); for (let iy = y0; iy <= y1; iy++) if (this.solid(ix, iy)) { p.x = ix - p.w / 2 - 1e-4; p.vx = 0; p.blockedT += STEP; return; } }
      else if (p.vx < 0) { const ix = Math.floor(p.x - p.w / 2); for (let iy = y0; iy <= y1; iy++) if (this.solid(ix, iy)) { p.x = ix + 1 + p.w / 2 + 1e-4; p.vx = 0; p.blockedT += STEP; return; } }
      p.blockedT = 0;
    }
    resolveY(p, wasGround) {
      const [x0, x1] = this.overlapX(p);
      if (p.vy <= 0) {
        // floor: solid tiles, one-way ledges (only from above), platforms (only from above)
        let best = null;
        const iy = Math.floor(p.y);
        for (let ix = x0; ix <= x1; ix++) {
          for (const ty of [iy, iy + 1]) {
            const top = ty + 1;
            if ((this.solid(ix, ty) || (this.oneWay(ix, ty) && p.prevBottom >= top - 0.06)) && p.y <= top && p.prevBottom >= top - 0.35 - Math.abs(p.vy) * STEP) { if (best == null || top > best) best = top; }
          }
        }
        let plat = null;
        for (const m of this.movers) {
          if (m.state === 'gone') continue;
          const top = m.y + m.h;
          if (p.x + p.w / 2 > m.x + 0.05 && p.x - p.w / 2 < m.x + m.w - 0.05 && p.y <= top + 0.01 && p.prevBottom >= top - 0.06 - Math.abs(p.vy) * STEP + Math.min(0, m.dy)) { if (best == null || top >= best) { best = top; plat = m; } }
        }
        if (best != null) { p.y = best; p.vy = 0; p.onGround = true; p.ground = plat; if (plat && plat.kind === 'fall' && plat.state === 'idle') { plat.state = 'shake'; plat.fallT = 0; this.onEvent('plank', plat); } return; }
        // walked off an edge?
        if (wasGround) { p.onGround = false; p.ground = null; } else p.onGround = false;
      } else {
        p.onGround = false; p.ground = null;
        const head = p.y + p.h, iy = Math.floor(head);
        let hits = [];
        for (let ix = x0; ix <= x1; ix++) {
          const k = this.tile(ix, iy), b = this.blockAt(ix, iy);
          if (this.solid(ix, iy) || (b && b.hidden && !b.used && p.prevBottom + p.h <= iy + 0.02)) { const ov = Math.min(p.x + p.w / 2, ix + 1) - Math.max(p.x - p.w / 2, ix); hits.push({ ix, iy, ov, k, b }); }
        }
        if (hits.length) {
          // corner correction: graze the corner of plain scenery → slide past instead of bonking
          if (hits.length === 1 && hits[0].ov < 0.2 && !L.INTERACTIVE.has(hits[0].k) && !(hits[0].b && hits[0].b.hidden)) {
            const hh = hits[0], nx = (p.x < hh.ix + 0.5) ? hh.ix - p.w / 2 - 0.01 : hh.ix + 1 + p.w / 2 + 0.01;
            if (!this.solid(Math.floor(nx), iy)) { p.x = nx; this.clampGate(p); return; }
          }
          hits.sort((a, b) => b.ov - a.ov);
          const hit = hits[0];
          p.y = hit.iy - p.h - 1e-4; p.vy = Math.min(0, -0.5); p.jumping = false; p.bumpT = 0;
          this.bump(hit.ix, hit.iy);
        }
      }
    }
    // ---------------------------------------------------------------- blocks
    bump(ix, iy) {
      const b = this.blockAt(ix, iy), p = this.player;
      this.onEvent('headbonk', { x: ix + 0.5, y: iy, block: !!b });
      if (!b) return;
      if (b.hidden && !b.used) { b.hidden = false; this.setTile(ix, iy, K.ITEMBLOCK); b.kind = K.ITEMBLOCK; }
      b.bumpT = 0;
      // enemies standing on the bumped block are knocked out
      for (const e of this.enemies) if (e.alive && Math.abs(e.x - (ix + 0.5)) < 0.5 + e.w / 2 && Math.abs(e.y - (iy + 1)) < 0.12) this.killEnemy(e, 'bump');
      if (b.kind === K.QBLOCK && b.slot != null) {
        const s = this.slots[b.slot];
        if (s.active && !s.solved) { s.tries++; this.onEvent('question', { slot: s.index, qIndex: s.qIndex, x: ix + 0.5, y: iy }); this.frozen = true; return; }
        this.onEvent('bump', { x: ix + 0.5, y: iy, kind: 'used' }); return;
      }
      if (b.kind === K.BRICK) {
        this.setTile(ix, iy, K.EMPTY); this.blocks.delete(ix + ',' + iy); this.stats.bricks++; this.stats.score += 50;
        this.onEvent('brick', { x: ix + 0.5, y: iy + 0.5 }); return;
      }
      if (b.kind === K.COINBLOCK && b.coins > 0) {
        b.coins--; this.addCoin(ix + 0.5, iy + 1, true);
        if (b.coins <= 0) { b.kind = K.USED; this.setTile(ix, iy, K.USED); b.used = true; }
        this.onEvent('bump', { x: ix + 0.5, y: iy, kind: 'coin' }); return;
      }
      if (b.kind === K.ITEMBLOCK && !b.used) {
        b.used = true; b.kind = K.USED; this.setTile(ix, iy, K.USED);
        const item = { kind: b.item || 'heart', x: ix + 0.5, y: iy + 0.02, y0: iy, vx: 0, vy: 0, rise: 0, gone: false, w: 0.6, h: 0.6, dir: p.x <= ix + 0.5 ? 1 : -1 };
        this.items.push(item); this.onEvent('item', item); return;
      }
      this.onEvent('bump', { x: ix + 0.5, y: iy, kind: 'used' });
    }
    addCoin(x, y, popped) { this.stats.coins++; this.stats.score += 10; this.onEvent('coin', { x, y, popped: !!popped, total: this.stats.coins }); }
    solveSlot(index) { // called by the game ONLY after a correct answer
      const s = this.slots[index]; if (!s || s.solved) return false;
      s.solved = true; const b = this.blockAt(s.x, s.y); if (b) { b.kind = K.USED; b.used = true; b.solved = true; b.bumpT = 0; }
      this.setTile(s.x, s.y, K.USED); this.frozen = false; this.gatePushT = 0; this.auto.waitT = -0.6;
      this.onEvent('gate-open', { slot: index, x: s.gateX });
      return true;
    }
    resumeAfterWrong() { /* the question stays open; nothing in the world moves — kept for symmetry */ }
    // ---------------------------------------------------------------- items
    stepItem(it, h) {
      if (it.rise < 1) { it.rise = Math.min(1, it.rise + h / 0.55); it.y = it.y0 + it.rise * 1.0; if (it.rise >= 1) { it.vx = it.dir * 2.2; } return; }
      it.vy = Math.max(-12, it.vy - 30 * h); it.x += it.vx * h;
      const ix = Math.floor(it.x + sign(it.vx) * it.w / 2); if (this.solid(ix, Math.floor(it.y + 0.3))) { it.vx = -it.vx; it.x += it.vx * h * 2; }
      it.y += it.vy * h; const fy = Math.floor(it.y);
      if (it.vy <= 0 && (this.solid(Math.floor(it.x), fy) || this.oneWay(Math.floor(it.x), fy))) { it.y = fy + 1; it.vy = 0; }
      const g = this.activeGate(); if (g && it.x > g.gateX - 0.3) { it.x = g.gateX - 0.3; it.vx = -Math.abs(it.vx); }
      if (it.y < L.Y_MIN) it.gone = true;
    }
    // ---------------------------------------------------------------- enemies
    inSafe(x) { for (const s of this.slots) if (s.active && !s.solved && x > s.x - 8 && x < s.x + 7) return true; return false; }
    stepEnemy(e, h) {
      if (!e.alive) { if (e.deadT >= 0) { e.deadT += h; e.vy -= 30 * h; e.y += e.vy * h * (e.squash ? 0 : 1); } return; }
      e.phase += h;
      if (e.fly) { // gliding patrol over [x0 − range, x0 + range]
        const r = Math.max(0.5, e.range), w = e.speed / r;
        const nx = e.x0 + r * Math.sin(e.phase * w); e.dir = sign(nx - e.x) || e.dir; e.x = nx; e.y = e.y0 + 0.35 * Math.sin(e.phase * w * 2.0);
        return;
      }
      if (e.def.speed === 0) return;
      e.vx = e.dir * e.speed;
      if (e.def.hop && e.onGround) { e.hopT -= h; if (e.hopT < 0) { e.vy = 6.5; e.onGround = false; e.hopT = 1.4; } }
      const nx = e.x + e.vx * h, front = Math.floor(nx + e.dir * e.w / 2), iy = Math.floor(e.y + 0.1);
      let turn = this.solid(front, iy) || (e.range && Math.abs(nx - e.x0) > e.range) || this.inSafe(nx + e.dir * e.w / 2);
      if (!turn && e.onGround) { const below = Math.floor(e.y - 0.5); if (!this.solid(front, below) && !this.oneWay(front, below)) turn = true; }
      if (e.def.hop && !e.onGround) turn = turn && this.solid(front, iy);   // hoppers keep their heading in the air unless they hit a wall
      if (turn) e.dir = -e.dir; else e.x = nx;
      e.vy = Math.max(-15, e.vy - 30 * h); e.y += e.vy * h;
      const fy = Math.floor(e.y);
      if (e.vy <= 0 && (this.solid(Math.floor(e.x), fy) || this.oneWay(Math.floor(e.x), fy))) { e.y = fy + 1; e.vy = 0; e.onGround = true; } else if (e.vy < -0.5) e.onGround = false;
      if (e.y < L.Y_MIN) { e.alive = false; e.deadT = -1; }
    }
    killEnemy(e, how) { e.alive = false; e.deadT = 0; e.squash = how === 'stomp'; e.vy = how === 'stomp' ? 0 : 6; this.stats.score += e.def.score; this.onEvent('enemy-down', { id: e.id, x: e.x, y: e.y, how, type: e.type }); }
    stepMover(m, h) {
      m.t += h; const ox = m.x, oy = m.y;
      if (m.kind === 'fall') {
        if (m.state === 'shake') { m.fallT += h; m.x = m.x0 + Math.sin(m.fallT * 60) * 0.03; if (m.fallT > 0.5) { m.state = 'falling'; m.vy = 0; this.onEvent('plank-fall', m); } }
        else if (m.state === 'falling') { m.vy -= 22 * h; m.y += m.vy * h; m.x = m.x0; if (m.y < L.Y_MIN - 1) { m.state = 'gone'; m.respawnT = 3.5; } }
        else if (m.state === 'gone') { m.respawnT -= h; if (m.respawnT <= 0) { m.state = 'idle'; m.x = m.x0; m.y = m.y0; } }
      } else {
        const w = 2 * m.speed / Math.max(0.5, m.range), off = m.range * (1 - Math.cos(m.t * w + m.phase)) / 2;   // peak speed = m.speed
        if (m.axis === 'x') m.x = m.x0 + off; else m.y = m.y0 + off;
        if (m.kind === 'boat') m.y = m.y0 + Math.sin(m.t * 2.1) * 0.04;
      }
      m.dx = m.x - ox; m.dy = m.y - oy;
    }
    // ---------------------------------------------------------------- interactions
    interact() {
      const p = this.player; if (p.dead) return;
      const px0 = p.x - p.w / 2, px1 = p.x + p.w / 2, py0 = p.y, py1 = p.y + p.h;
      // coins
      for (const c of this.coins) if (!c.taken && c.x > px0 - 0.3 && c.x < px1 + 0.3 && c.y > py0 - 0.3 && c.y < py1 + 0.3) { c.taken = true; this.addCoin(c.x, c.y, false); }
      // items
      for (const it of this.items) if (!it.gone && it.rise >= 1 && Math.abs(it.x - p.x) < (it.w + p.w) / 2 && it.y < py1 && it.y + it.h > py0) { it.gone = true; if (it.kind === 'heart') { p.hearts = Math.min(this.opts.hearts + 2, p.hearts + 1); } this.stats.score += 500; this.onEvent('collect', { kind: it.kind, x: it.x, y: it.y }); }
      // enemies
      for (const e of this.enemies) {
        if (!e.alive) continue;
        const ex0 = e.x - e.w / 2, ex1 = e.x + e.w / 2, ey0 = e.y, ey1 = e.y + e.h;
        if (px1 <= ex0 + 0.06 || px0 >= ex1 - 0.06 || py1 <= ey0 + 0.04 || py0 >= ey1) continue;
        const fromAbove = p.vy < 0 && p.prevBottom >= ey1 - 0.16;
        if (fromAbove && e.def.stomp) {
          this.killEnemy(e, 'stomp'); this.chain++; this.stats.stomps++;
          p.vy = this.input.jump || (this.opts.autorun && this.input.jump) ? P.stompHeldV : P.stompV; p.jumping = true; p.holdT = P.holdMax; p.y = ey1; p.tuck = 1; p.airJumps = this.opts.doubleJump ? 1 : 0;
          this.stats.score += 100 * (this.chain - 1);
          this.onEvent('stomp', { x: e.x, y: ey1, chain: this.chain, type: e.type });
        } else if (fromAbove && !e.def.stomp) { this.hurt(e, true); }
        else this.hurt(e, false);
      }
    }
    hurt(e, bounce) {
      const p = this.player; if (p.invulT > 0 || p.dead) { if (bounce) p.vy = Math.max(p.vy, 6); return; }
      p.hearts--; p.invulT = P.invul; p.hurtT = P.hurtCtl; const away = sign(p.x - e.x) || -p.facing;
      p.vx = away * P.hurtVX; p.vy = bounce ? 8 : P.hurtV; p.onGround = false; p.ground = null;
      this.onEvent('hurt', { hearts: p.hearts, x: p.x, y: p.y, by: e.type });
      if (p.hearts <= 0) { p.dead = true; this.onEvent('die', { cause: 'hearts', x: p.x }); }
    }
    // ---------------------------------------------------------------- respawn after a lost life (solved questions stay solved)
    respawnPlayer() {
      const r = this.respawn; const hearts = this.opts.hearts;
      this.player = this.newPlayer(r.x, r.y); this.player.hearts = hearts; this.player.invulT = 1.2;
      // put enemies back (only those not too close to the respawn point), restore falling planks
      this.resetEnemies(); for (const e of this.enemies) if (Math.abs(e.x - r.x) < 6) { e.alive = false; e.deadT = -1; }
      for (const m of this.movers) { const n = this.makeMover(this.L.movers[m.id], m.id); Object.assign(m, n); }
      this.items = []; this.frozen = false; this.gatePushT = 0; this.auto = { state: 'run', jumpT: 0, waitT: 0 };
      this.onEvent('respawn', r);
    }
    // ---------------------------------------------------------------- snapshot for tests / debug
    snapshot() { const p = this.player; return { x: p.x, y: p.y, vx: p.vx, vy: p.vy, onGround: p.onGround, hearts: p.hearts, dead: p.dead, frozen: this.frozen, gate: this.activeGate() ? this.activeGate().index : -1, coins: this.stats.coins, score: this.stats.score, time: this.time }; }
  }
  root.QHEngine = { Engine, P, STEP, ENEMY, SOLID };
})(typeof window !== 'undefined' ? window : globalThis);
</script>

        const c = o.color ? (Array.isArray(o.color) ? o.color[(Math.random() * o.color.length) | 0] : o.color) : new T.Color(1, 1, 1);
        this.col[i * 3] = c.r; this.col[i * 3 + 1] = c.g; this.col[i * 3 + 2] = c.b;
        this.max[i] = this.life[i] = (o.life || 0.8) * (0.6 + Math.random() * 0.4); this.size[i] = (o.size || 0.3) * (0.6 + Math.random() * 0.6);
      }
    }
    update(dt) {
      for (let i = 0; i < this.n; i++) {
        if (this.life[i] <= 0) { if (this.size[i]) this.size[i] = 0; continue; }
        this.life[i] -= dt; const k = Math.max(0, this.life[i] / this.max[i]);
        this.v[i * 3 + 1] += this.grav * dt; const d = Math.exp(-this.drag * dt); this.v[i * 3] *= d; this.v[i * 3 + 1] *= d; this.v[i * 3 + 2] *= d;
        this.p[i * 3] += this.v[i * 3] * dt; this.p[i * 3 + 1] += this.v[i * 3 + 1] * dt; this.p[i * 3 + 2] += this.v[i * 3 + 2] * dt;
        this.size[i] = Math.abs(this.size[i]) * (k > 0.02 ? 1 : 0) * (0.985 + 0.015 * k);
        if (this.life[i] <= 0) this.size[i] = 0;
      }
      this.g.attributes.position.needsUpdate = true; this.g.attributes.size.needsUpdate = true; this.g.attributes.color.needsUpdate = true;
    }
  }

  class Bits {
    constructor(parent, n, geo, mat) {
      this.n = n; this.mesh = new T.InstancedMesh(geo, mat, n); this.mesh.frustumCulled = false; this.mesh.castShadow = false; this.mesh.instanceMatrix.setUsage(T.DynamicDrawUsage);
      this.s = Array.from({ length: n }, () => ({ life: 0, x: 0, y: 0, z: 0, vx: 0, vy: 0, vz: 0, rx: 0, ry: 0, rz: 0, wx: 0, wy: 0, wz: 0, sc: 1, flut: 0 }));
      const zero = new T.Matrix4().makeScale(0, 0, 0); for (let i = 0; i < n; i++) this.mesh.setMatrixAt(i, zero);
      if (mat.vertexColors || this.colored) { /* noop */ }
      this.mesh.instanceColor = new T.InstancedBufferAttribute(new Float32Array(n * 3).fill(1), 3);
      parent.add(this.mesh); this.next = 0;
    }
    emit(x, y, z, k, o = {}) {
      for (let j = 0; j < k; j++) {
        const i = this.next; this.next = (this.next + 1) % this.n; const s = this.s[i];
        s.life = (o.life || 1.2) * (0.7 + Math.random() * 0.5); s.x = x + (Math.random() - 0.5) * (o.spread || 0.5); s.y = y + (Math.random() - 0.5) * (o.spread || 0.5); s.z = z + (Math.random() - 0.5) * (o.spreadZ || 0.4);
        const a = (o.angle || 0) + (Math.random() - 0.5) * (o.cone || Math.PI), sp = (o.speed || 5) * (0.5 + Math.random() * 0.6);
        s.vx = Math.sin(a) * sp; s.vy = Math.cos(a) * sp + (o.vy || 0); s.vz = (Math.random() - 0.5) * sp * 0.4;
        s.rx = Math.random() * 6; s.ry = Math.random() * 6; s.rz = Math.random() * 6; s.wx = (Math.random() - 0.5) * 14; s.wy = (Math.random() - 0.5) * 14; s.wz = (Math.random() - 0.5) * 14;
        s.sc = (o.size || 1) * (0.7 + Math.random() * 0.6); s.g = o.gravity == null ? -18 : o.gravity; s.flut = o.flutter || 0;
        if (o.colors) { const c = o.colors[(Math.random() * o.colors.length) | 0]; this.mesh.instanceColor.setXYZ(i, c.r, c.g, c.b); this.mesh.instanceColor.needsUpdate = true; }
      }
    }
    update(dt) {
      for (let i = 0; i < this.n; i++) {
        const s = this.s[i]; if (s.life <= 0) continue; s.life -= dt;
        s.vy += s.g * dt; if (s.flut) { s.vx *= Math.exp(-2.5 * dt); s.vy = Math.max(s.vy, -1.6); s.vx += Math.sin(s.life * 7 + i) * s.flut * dt; }
        s.x += s.vx * dt; s.y += s.vy * dt; s.z += s.vz * dt; s.rx += s.wx * dt; s.ry += s.wy * dt; s.rz += s.wz * dt;
        const sc = s.life > 0 ? s.sc * Math.min(1, s.life * 3) : 0;
        tmpQ.setFromEuler(tmpE.set(s.rx, s.ry, s.rz)); tmpS.set(sc, sc, sc); tmpP.set(s.x, s.y, s.z); tmpM.compose(tmpP, tmpQ, tmpS); this.mesh.setMatrixAt(i, tmpM);
      }
      this.mesh.instanceMatrix.needsUpdate = true;
    }
  }

  function boxes(v, mat, list, o = {}) {
    if (!list.length) return null; const g = new T.BoxGeometry(1, 1, 1); g.translate(0, 0.5, 0);
    const m = new T.InstancedMesh(g, mat, list.length); m.castShadow = o.shadow !== false && v.q !== 'low'; m.receiveShadow = true; m.frustumCulled = false;
    list.forEach((b, i) => m.setMatrixAt(i, mtx(b.x, b.y, b.z, b.ry || 0, b.w, b.h, b.d)));
    if (o.colors) { list.forEach((b, i) => m.setColorAt(i, b.c || new T.Color(1, 1, 1))); }
    (o.parent || v.stage).add(m); return m;
  }

  function treeDef(trunkCol = '#6b4f36', leafCol = '#9fbf8a', h = 4.2, spread = 1.0) {
    const trunk = new T.CylinderGeometry(0.12, 0.2, h * 0.6, 8); trunk.translate(0, h * 0.3, 0);
    const blobs = []; for (const [x, y, z, r] of [[0, 0.78, 0, 1.0], [0.6, 0.7, 0.2, 0.75], [-0.55, 0.68, -0.15, 0.8], [0.1, 0.92, -0.35, 0.7], [-0.15, 0.6, 0.45, 0.65]]) { const g = new T.SphereGeometry(r * spread, 18, 12); const p = g.attributes.position; for (let i = 0; i < p.count; i++) { const k = 1 + 0.1 * Math.sin(p.getX(i) * 9 + p.getZ(i) * 5) * Math.cos(p.getY(i) * 7); p.setXYZ(i, p.getX(i) * k + x * spread, p.getY(i) * k * 0.8 + y * h, p.getZ(i) * k + z * spread); } g.computeVertexNormals(); blobs.push(g); }
    return { parts: [{ name: 'trunk', geo: trunk, mat: std({ color: C(trunkCol), roughness: 0.95, map: X.photo('trunk_c') }) }, { name: 'leaves', geo: P.merge(blobs), mat: std({ normalMap: X.photo('hedge_n', { srgb: false, repeat: [3, 3] }), normalScale: new T.Vector2(1.6, 1.6), color: C(leafCol), roughness: 0.8 }) }] };
  }
